cmake_minimum_required(VERSION 3.16)
project(structured_basic C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(sbasic_core src/compiler.c src/vm.c)
target_include_directories(sbasic_core PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(sbasic_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "avr")
  add_executable(sbasic_firmware platform/avr.c)
  target_link_libraries(sbasic_firmware PRIVATE sbasic_core)
else()
  add_executable(sbasic platform/host.c)
  target_link_libraries(sbasic PRIVATE sbasic_core)
  add_executable(api_tests tests/api.c)
  target_link_libraries(api_tests PRIVATE sbasic_core)
  enable_testing()
  add_test(NAME api COMMAND api_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME acceptance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/acceptance.py $<TARGET_FILE:sbasic>)
  endif()
endif()
