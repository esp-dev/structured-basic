cmake_minimum_required(VERSION 3.17)
project(structured_basic VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(SB_PROFILE "DEBUG" CACHE STRING "DEBUG, SANITIZED, EXTENDED, RELEASE")
set_property(CACHE SB_PROFILE PROPERTY STRINGS DEBUG SANITIZED EXTENDED RELEASE)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "avr")
  if(SB_PROFILE STREQUAL "RELEASE")
    add_compile_options(-Os -g -DNDEBUG -flto -ffunction-sections -fdata-sections)
    add_link_options(-flto -Wl,--gc-sections)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang")
      add_link_options(-fuse-ld=lld)
    endif()
  else()
    add_compile_options(-g -UNDEBUG -DSB_DEBUG)
    if(SB_PROFILE MATCHES "SANITIZED|EXTENDED")
      if(SB_PROFILE STREQUAL "EXTENDED")
        set(SB_SANITIZERS "address,undefined,integer,local-bounds")
      else()
        set(SB_SANITIZERS "address,undefined")
      endif()
      add_compile_options(-O1 -fno-omit-frame-pointer -fno-sanitize-recover=all "-fsanitize=${SB_SANITIZERS}" -DSB_SANITIZED)
      add_link_options(-fno-sanitize-recover=all "-fsanitize=${SB_SANITIZERS}")
      set_source_files_properties(tests/api.c PROPERTIES COMPILE_OPTIONS "-fno-sanitize=unsigned-integer-overflow")
    else()
      add_compile_options(-O0)
    endif()
  endif()
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fno-common)
include(cmake/SBasicCore.cmake)
sbasic_add_core(sbasic_core WITH_TERMINAL)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "avr")
  add_executable(sbasic_firmware platform/avr.c)
  target_link_libraries(sbasic_firmware PRIVATE sbasic_core -Wl,-u,vfprintf printf_flt)
else()
  add_executable(sbasic platform/host.c platform/mock.c)
  target_link_libraries(sbasic PRIVATE sbasic_core)
  if(SB_PROFILE STREQUAL "RELEASE")
    target_link_options(sbasic PRIVATE "-Wl,--verbose,-Map=${CMAKE_CURRENT_BINARY_DIR}/sbasic.map")
    find_program(SB_STRIP NAMES llvm-strip strip REQUIRED)
    add_custom_command(TARGET sbasic POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:sbasic> "${CMAKE_CURRENT_BINARY_DIR}/sbasic-unstripped${CMAKE_EXECUTABLE_SUFFIX}"
      COMMAND ${SB_STRIP} --strip-all $<TARGET_FILE:sbasic>)
  endif()
  add_executable(api_tests tests/api.c)
  target_link_libraries(api_tests PRIVATE sbasic_core)
  add_executable(stress_tests tests/stress10.c platform/mock.c)
  target_include_directories(stress_tests PRIVATE platform)
  target_link_libraries(stress_tests PRIVATE sbasic_core)
  enable_testing()
  add_test(NAME api COMMAND api_tests)
  add_test(NAME stress COMMAND stress_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME acceptance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/acceptance.py $<TARGET_FILE:sbasic>)
  add_test(NAME basic10 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/extended.py $<TARGET_FILE:sbasic>)
  add_test(NAME grammar COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/grammar.py $<TARGET_FILE:sbasic>)
endif()
